package main

import "fmt"

// 1. Declaración de constantes globales para propagación de constantes
const PI = 3.1416
const GRAVEDAD = 9.81
const TASA_IVA = 0.16
const FACTOR_ESCALA = 4
const BONO_BASE = 250

// 2. Función con parámetros float64, expresiones aritméticas y retorno
func calcularEnergiaMecanica(masa float64, altura float64, velocidad float64) float64 {
    // Expresión de energía potencial
    var energiaPotencial float64
    energiaPotencial = masa * GRAVEDAD * altura

    // Expresión de energía cinética con paréntesis y potencia
    var energiaCinetica float64
    energiaCinetica = (masa * (velocidad ^ 2)) / 2.0

    // Suma de energías con ajuste constante
    var energiaTotal float64
    energiaTotal = energiaPotencial + energiaCinetica + 12.5 * 2.0

    fmt.Println("Energia calculada:", energiaTotal)
    return energiaTotal
}

// 3. Función con parámetros int, condicionales y bucle for
func procesarLoteProduccion(unidadesBase int, turnos int) int {
    var metaMinima int
    metaMinima = 100

    // Expresión con múltiples niveles de precedencia
    var produccionBruta int
    produccionBruta = unidadesBase * turnos + BONO_BASE / 5 - 10

    var mermaEstimada int
    mermaEstimada = (unidadesBase + 20) / FACTOR_ESCALA

    var produccionNeta int
    produccionNeta = produccionBruta - mermaEstimada

    if produccionNeta >= metaMinima {
        fmt.Println("Meta de produccion alcanzada:", produccionNeta)
    } else {
        fmt.Println("Produccion por debajo de la meta:", produccionNeta)
    }

    // Ciclo for para acumular ajuste por turno
    var iterador int
    iterador = 1
    var acumuladoTurno int
    acumuladoTurno = 0

    for iterador <= turnos {
        acumuladoTurno = acumuladoTurno + iterador * FACTOR_ESCALA
        iterador = iterador + 1
    }

    var resultadoFinal int
    resultadoFinal = produccionNeta + acumuladoTurno
    return resultadoFinal
}

// 4. Función booleana para evaluar condiciones relacionales y lógicas
func validarEstadoSistema(temperatura float64, presion float64, activo bool) bool {
    var tempMaxima float64
    tempMaxima = 95.5
    var presionMaxima float64
    presionMaxima = 120.0

    var tempSegura bool
    tempSegura = temperatura < tempMaxima

    var presionSegura bool
    presionSegura = presion <= presionMaxima

    var estadoOperativo bool
    estadoOperativo = tempSegura && presionSegura && activo

    if !estadoOperativo {
        fmt.Println("Alerta: Sistema fuera de parametros seguros")
    } else {
        fmt.Println("Sistema operando dentro de parametros normales")
    }

    return estadoOperativo
}

// 5. Función principal con bloque de optimización y código intermedio
func main() {
    // Mensaje de inicio del sistema
    var tituloSistema string
    tituloSistema = "=== INICIANDO PRUEBA ==="
    fmt.Println(tituloSistema)

    // SECCIÓN A: Variables constantes candidatas a Propagación y Eliminación
    A := 10
    B := 5
    C := 2
    D := 4
    E := 2
    F := 1
    G := 3
    H := 6

    // SECCIÓN B: Expresiones para plegado de constantes (Constant Folding)
    plegado1 := 20 + 15 * 4 - 10
    plegado2 := 100 / 5 + 8 * 3 - 4
    plegado3 := 50 * 2 - 30 / 3 + 15

    fmt.Println("Plegado 1:", plegado1)
    fmt.Println("Plegado 2:", plegado2)
    fmt.Println("Plegado 3:", plegado3)

    // Operaciones encadenadas para propagación en cascada y código muerto
    pasoIntermedio1 := A + B * C
    pasoIntermedio2 := pasoIntermedio1 * D - E
    resultadoCascada := pasoIntermedio2 + G * H

    fmt.Println("Resultado en cascada:", resultadoCascada)

    // SECCIÓN C: Expresiones complejas (Cuádruplos, Prefija, Posfija)
    // Expresión 1: Precedencia básica y paréntesis
    exp1 := (A + B) * C - D
    fmt.Println("Exp 1:", exp1)

    // Expresión 2: División y multiplicación combinadas
    exp2 := A * B + C / D
    fmt.Println("Exp 2:", exp2)

    // Expresión 3: Agrupaciones dobles
    exp3 := (A + B) * (C - D)
    fmt.Println("Exp 3:", exp3)

    // Expresión 4: Subexpresión entre paréntesis multiplicada
    exp4 := A + (B - C) * D
    fmt.Println("Exp 4:", exp4)

    // Expresión 5: División de agrupación más desplazamiento
    exp5 := (A - B) / C + D
    fmt.Println("Exp 5:", exp5)

    // Expresión 6: Combinación de suma agrupada y cociente
    exp6 := A * (B + C) - D / E
    fmt.Println("Exp 6:", exp6)

    // Expresión 7: Cociente de dos agrupaciones
    exp7 := (A + B) / (C - D) + E
    fmt.Println("Exp 7:", exp7)

    // Expresión 8: Multiplicación y división sobre agrupación
    exp8 := A + B * (C - D) / E
    fmt.Println("Exp 8:", exp8)

    // Expresión 9: Tres bloques agrupados con multiplicación y división
    exp9 := (A - B) * (C + D) / (E - F)
    fmt.Println("Exp 9:", exp9)

    // Expresión 10: Potencia, paréntesis y múltiples operadores
    exp10 := A + B * (C ^ D - E) / (F + G) - H
    fmt.Println("Exp 10:", exp10)

    // Expresión 11: Potencia de dos subexpresiones agrupadas
    exp11 := ((A + B) / C) ^ (D - E)
    fmt.Println("Exp 11:", exp11)

    // Expresión 12: Potencia con producto de binomios en la base
    exp12 := ((A - B) * (C + D)) ^ (E / F)
    fmt.Println("Exp 12:", exp12)

    // SECCIÓN D: Invocación de funciones, ámbitos y tipos de retorno
    var energiaCalculada float64
    energiaCalculada = calcularEnergiaMecanica(12.5, 8.0, 4.0)
    fmt.Println("Retorno Energia:", energiaCalculada)

    var totalProduccion int
    totalProduccion = procesarLoteProduccion(45, 3)
    fmt.Println("Retorno Produccion:", totalProduccion)

    var sistemaOk bool
    sistemaOk = validarEstadoSistema(78.4, 105.0, true)
    fmt.Println("Estado del sistema:", sistemaOk)

    // Cálculo financiero final usando constantes globales
    var subtotal float64
    subtotal = 1500.0 + 250.0 * 2.0
    var totalConImpuesto float64
    totalConImpuesto = subtotal + subtotal * TASA_IVA
    fmt.Println("Total con IVA:", totalConImpuesto)

    var cierreSistema string
    cierreSistema = "=== EJECUCION Y ANALISIS COMPLETADOS ==="
    fmt.Println(cierreSistema)
}
